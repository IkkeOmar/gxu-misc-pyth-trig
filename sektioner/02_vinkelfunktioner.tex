\begin{regelbox}
\textbf{Regel --- sinus, cosinus og tangens:} For en retvinklet trekant
med vinkel $\theta$ (i grader), katet $a$ over for $\theta$, katet $b$
hos $\theta$ og hypotenuse $c$:
\begin{equation*}
\sin\theta=\frac{a}{c}\qquad\cos\theta=\frac{b}{c}\qquad\tan\theta=\frac{a}{b}
\end{equation*}
\end{regelbox}

\subsection*{Eksempel --- $a=5$, $c=13$}

\noindent
Først findes $b$ med Pythagoras:
\begin{align*}
  b &= \sqrt{13^{2}-5^{2}} = \sqrt{169-25} = \sqrt{144} = 12
\end{align*}
Derefter de tre forhold:
\begin{align*}
  \sin\theta &= \frac{a}{c} = \frac{5}{13}, &
  \cos\theta &= \frac{b}{c} = \frac{12}{13}, &
  \tan\theta &= \frac{a}{b} = \frac{5}{12}
\end{align*}
