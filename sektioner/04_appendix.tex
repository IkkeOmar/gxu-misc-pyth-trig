\section*{Appendix --- Facit}

\noindent
\textbf{Type A --- Pythagoras}
\begin{itemize}[leftmargin=2em, itemsep=4pt]
  \item \textbf{1.1:} $c=\sqrt{6^{2}+8^{2}}=\sqrt{36+64}=\sqrt{100}=10$.
  \item \textbf{1.2:} $b=\sqrt{13^{2}-5^{2}}=\sqrt{169-25}=\sqrt{144}=12$.
  \item \textbf{1.3:} $a=\sqrt{15^{2}-9^{2}}=\sqrt{225-81}=\sqrt{144}=12$.
  \item \textbf{1.4:} $c=\sqrt{20^{2}+21^{2}}=\sqrt{400+441}=\sqrt{841}=29$.
\end{itemize}

\noindent
\textbf{Type B --- Trigonometri}
\begin{itemize}[leftmargin=2em, itemsep=4pt]
  \item \textbf{1.5:} $a=c\sin\theta=10\cdot\tfrac{1}{2}=5$. \quad
        $b=c\cos\theta=10\cdot\tfrac{\sqrt{3}}{2}=5\sqrt{3}\approx 8{,}66$.
  \item \textbf{1.6:} $\tan60^{\circ}=\sqrt{3}\Rightarrow b=a/\tan\theta=7/\sqrt{3}\approx 4{,}04$. \quad
        $c=a/\sin\theta=7/(\sqrt{3}/2)=14/\sqrt{3}\approx 8{,}08$.
  \item \textbf{1.7:} $a=b=c\sin45^{\circ}=12\cdot\tfrac{\sqrt{2}}{2}=6\sqrt{2}\approx 8{,}49$. \quad
        $b=a\approx 8{,}49$.
  \item \textbf{1.8:} $\tan53{,}13^{\circ}=4/3\Rightarrow a=b\tan\theta=12\cdot\tfrac{4}{3}=16$. \quad
        $c=\sqrt{a^{2}+b^{2}}=\sqrt{256+144}=\sqrt{400}=20$.
\end{itemize}

\noindent
\textbf{Type C --- Beviser og aflæsning}
\begin{itemize}[leftmargin=2em, itemsep=4pt]
  \item \textbf{1.9:} $(a/c)^{2}+(b/c)^{2}=(a^{2}+b^{2})/c^{2}=c^{2}/c^{2}=1$ ved Pythagoras.
  \item \textbf{1.10:} $(\tfrac{3}{5})^{2}+(\tfrac{4}{5})^{2}=\tfrac{9}{25}+\tfrac{16}{25}=\tfrac{25}{25}=1$ $\checkmark$.
\end{itemize}
