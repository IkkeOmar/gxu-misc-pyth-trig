\begin{regelbox}
\textbf{Regel --- enhedscirklen:} En cirkel med radius 1. Punktet
$P=(\cos\theta,\sin\theta)$ ligger på cirklen. Når vinklen $\theta$ måles
mod $x$-aksen, er $\cos\theta$ projektionen på $x$-aksen og $\sin\theta$
projektionen på $y$-aksen.
\end{regelbox}

\subsection*{Eksempel --- $(\tfrac{3}{5},\tfrac{4}{5})$ på enhedscirklen}

\noindent
Tjek grundrelationen:
\begin{align*}
  \cos^{2}\theta+\sin^{2}\theta = (\tfrac{3}{5})^{2}+(\tfrac{4}{5})^{2}
    = \tfrac{9}{25}+\tfrac{16}{25} = \tfrac{25}{25} = 1 \checkmark
\end{align*}

\note{Når $\cos^{2}\theta+\sin^{2}\theta=1$, ligger punktet på enhedscirklen.}

\subsection*{Eksempel --- aflæs $(\cos\theta,\sin\theta)$ for standardvinkler}

\noindent
$\theta=0^{\circ}$: punktet er $(1,0)$. $\quad$ $\theta=90^{\circ}$: punktet er $(0,1)$. $\quad$ $\theta=270^{\circ}$: punktet er $(0,-1)$.
